\section{আমর}

\noindent যে ফেল দিয়ে কাজ করার আদেশ বা অনুরোধ বোঝায়, তাকে
{\arabicfont \textarabic{فعل أمر}} বলে।
যেমন {\arabicfont \textarabic{اِفْعَلْ}} কর,
{\arabicfont \textarabic{اُنْصُرْ}} সাহায্য কর।

\vspace{0.3cm}

\noindent মুজারে মারুফ থেকে আমর হয়।

\begin{enumerate}
  \item হরফে মুজারা {\arabicfont \textarabic{ي ت أ ن}} ফেলো।
  \item আদ্যবর্ণ সাকিন থাকলে শুরুতে হামযায় ওয়াসল বসাও।
        আইনে যের বা যবর হলে হামযায় যের;
        আইনে পেশ হলে হামযায় পেশ।
  \item লাম সহীহ হলে সাকিন করো।
        লাম ইল্লাত ({\arabicfont \textarabic{ا و ي}}) হলে লাম ফেলো।
  \item নুনে ই'রাব ফেলো।
        গায়েব ও মুতাকাল্লিমে মুজারের আগে যেরযুক্ত লাম বসে:
        {\arabicfont \amrlim}।
\end{enumerate}
