% \aug: an AUGMENT letter (زائدة) — a non-root letter added by the bab
% pattern. Greyed so the black root letters stand out. Same coloured-run
% approach as \sfx, just grey; used inside a single \textarabic{}.
\newcommand{\aug}[1]{{\color{augmentgray}#1}}

% A colour change splits the word into separate shaping runs, so the
% letters either side of it stop joining (فَ ا عِلٌ). A ZERO WIDTH JOINER
% (U+200D) at the end of one run and the start of the next restores the
% cursive join. \zwj is the bare joiner; \zj{<letter>} is a joiner only
% if <letter> can join to the next one, so it is safe after a root
% letter passed as an argument (ا د ذ ر ز و … never join forward).
\newcommand{\zwj}{^^^^200d}
\newcommand{\nonjoiners}{اأإآدذرزوؤءة}
\makeatletter
\newcommand{\zj}[1]{%
  \edef\@tempa{\noexpand\in@{#1}{\nonjoiners}}\@tempa
  \ifin@\else\zwj\fi
}
\makeatother

% ─── Arabic words ───────────────────────────────────────────
% Each word is wrapped in \textarabic{} so the whole word is ONE
% right-to-left unit. Without this, the coloured root and suffix
% get reordered left-to-right and the red lands on the wrong side.
%
% SPECIAL CASE — when the marker (علامة) is a bare diacritic, not an
% added letter (e.g. the final fatha in فَعَلَ for واحد مذكر غائب):
% XeLaTeX cannot colour a diacritic differently from its base letter,
% because changing colour mid-cluster detaches the mark. Instead we
% OVERLAY: the fully-red word is drawn first (bottom layer, and it sets
% the box width), then a black copy LACKING the final mark is \llap-ed
% on top. The black letters hide the red ones; only the red diacritic
% shows through. Kept inside \mbox so it centres correctly in a cell.
% Usage: \redmark{full red word}{same word WITHOUT the final mark}
% (\ovl, further down, generalises this to several marks and colours.)
\newcommand{\redmark}[2]{\mbox{\textarabic{\sfx{#1}}\llap{\textarabic{\myroot{#2}}}}}

% ─── المضارع (present/future) ────────────────────────────────
% In المضارع the markers are PREFIXES (حروف المضارعة: يَ تَ أَ نَ) as
% well as suffixes. So both ends are red, the root فعل stays black.
%
% \mud{prefix}{black stem}{suffix}: a form whose marker suffix is an
% ADDED LETTER (انِ، ونَ، ينَ، نَ). Straightforward coloured runs.
\newcommand{\mud}[3]{\textarabic{\sfx{#1}\myroot{#2}\sfx{#3}}}
%
% \mudraf{prefix}: a singular form ending in a bare damma (raf'), e.g.
% يَفْعَلُ. The final damma is the marker but a bare diacritic, so —as
% with the ماضي fatha— we OVERLAY. Crucially BOTH layers split the
% prefix the same way, so they shape (and measure) identically and
% align. Bottom = red prefix + red stem+damma; top = red prefix +
% black stem (no damma). Black hides red stem; red damma shows.
\newcommand{\mudraf}[1]{%
  \mbox{\textarabic{\sfx{#1}\sfx{فْعَلُ}}\llap{\textarabic{\sfx{#1}\myroot{فْعَل}}}}%
}

% ─── باب الافتعال (Form VIII) مصدر ───────────────────────────
% The masdar pattern is الِافْتِعَال. The AUGMENT letters are grey: the
% article+hamzat-waṣl الِا, the infixed تِ, and the long ا. The three
% root letters stay black with their pattern vowels (sukūn, fatḥa, ḍamma).
% Pass the three bare root consonants: \iftmasdar{F1}{F2}{F3}
\newcommand{\iftmasdar}[3]{%
  \textarabic{\aug{الِا}\myroot{#1ْ}\aug{تِ}\myroot{#2َ}\aug{ا}\myroot{#3ُ}}%
}

% ─── باب التفعّل (Form V) مصدر ────────────────────────────────
% The masdar pattern is التَّفَعُّل. The leading augment التَّ (article ال
% + the prefixed ت, doubled by sun-letter assimilation of the lām) is
% grey; the rest of the word (the root letters with their pattern vowels
% and the doubling shadda) stays black. Pass the black remainder.
% \tafmasdar{<word without the leading التَّ>}
\newcommand{\tafmasdar}[1]{\textarabic{\aug{التَّ}\myroot{#1}}}

% ─── باب التفعيل (Form II) مصدر ───────────────────────────────
% The masdar pattern is التَّفْعِيل. Two augments are grey: the leading
% التَّ (article + prefixed ت) and the long ي of the تفعيل pattern. The
% three root letters stay black (sukūn on F1, kasra on F2, case vowel
% on F3). Pass the three bare root consonants: \tafeelmasdar{F1}{F2}{F3}
\newcommand{\tafeelmasdar}[3]{%
  \textarabic{\aug{التَّ}\myroot{#1ْ#2ِ}\aug{ي}\myroot{#3ُ}}%
}

% ─── اسم مشتق (derived nouns) ─────────────────────────────────
% Same scheme as the مصدر helpers: the three root letters stay black,
% the pattern's AUGMENT letters are grey, joined across each colour
% change with \zwj / \zj. Pass the three bare root consonants; with

% مزيد فيه forms, where the مضارع prefix (red) is swapped for مُ (red):
% \mazid{<red marker>}{<grey augments>}{<black root remainder>}
% e.g. \mazid{مُ}{سْتَ}{عْمِلٌ} → مُسْتَعْمِلٌ. Always joins across both
% colour changes, so the marker and augments must end in a joining
% letter (مُ يَ تَ نَ سْتَ are fine; أَ is not).
\newcommand{\mazid}[3]{\textarabic{\sfx{#1\zwj}\aug{\zwj#2\zwj}\myroot{\zwj#3}}}

% ─── اسم مشتق গঠন (construction from the مضارع) ──────────────
% The source مضارع: its علامة (red) is what gets dropped.
% \mudsrc{<red علامة>}{<black rest>}, e.g. \mudsrc{يَ}{نْصُرُ}
\newcommand{\mudsrc}[2]{\textarabic{\sfx{#1\zwj}\myroot{\zwj#2}}}
%
% The built word: a NEW or CHANGED haraka is red, a kept one stays with
% its letter's colour. A bare haraka can't be coloured on its own (see
% \redmark), so \ovl overlays two copies of ONE spec, written with:
%   \rt{…} a run of root letters (black)   \ad{…} a run of added letters (grey)
%   \nh{<mark>} a new/changed haraka — put it inside the run, after its letter
% Bottom copy: every run red, \nh marks kept. Top copy: runs black/grey,
% \nh marks dropped, so only they show through red. Both copies come from
% the same spec, so their colour runs split identically and the glyphs
% (incl. kerning across run edges, e.g. ر ى) line up exactly.
\makeatletter
% \mk stays red on both layers (a marker letter, not a bare haraka).
\newcommand{\ovl}[1]{\mbox{%
  \textarabic{\let\rt\sfx\let\ad\sfx\let\mk\sfx\let\nh\@firstofone#1}%
  \llap{\textarabic{\let\rt\myroot\let\ad\aug\let\mk\sfx\let\nh\@gobble#1}}}}
\makeatother
